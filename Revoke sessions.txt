<#
    Revoke-Sessions.ps1
    Day 7 - Programmatic threat containment.
    Revokes all sign-in sessions for a compromised user via Microsoft Graph,
    forcing re-authentication everywhere and evicting an attacker.

    A password reset alone does NOT kill live tokens - revoking sessions does.
    Requires the User.RevokeSessions.All scope (a write action - least privilege:
    read scopes cannot perform this).
#>

# Connect with the least-privilege write scope needed for this action
Connect-MgGraph -Scopes "User.RevokeSessions.All","User.Read.All"

# Target the compromised user
$UPN = "Bala@shubhamstreetgmail.onmicrosoft.com"

Write-Host "Revoking all sessions for $UPN ..."
Revoke-MgUserSignInSession -UserId $UPN
Write-Host "Done - $UPN must now re-authenticate everywhere."

# Real-world containment playbook: revoke sessions + reset password (+ disable if needed)
